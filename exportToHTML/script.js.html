script.js 
// Import the questions from the external file
import allQuestions from './questions.js';

document.addEventListener('DOMContentLoaded', () => {
    // Settings and UI state
    let isSpacedRepetition = false; // false = elimination mode, true = spaced repetition mode
    let isDarkMode = false; // Theme state

    // DOM elements
    const startScreen = document.getElementById('startScreen');
    const quizScreen = document.getElementById('quizScreen');
    const resultsScreen = document.getElementById('resultsScreen');
    const settingsOverlay = document.getElementById('settingsOverlay');

    const startBtn = document.getElementById('startBtn');
    const nextBtn = document.getElementById('nextBtn');
    const reviewBtn = document.getElementById('reviewBtn');
    const restartBtn = document.getElementById('restartBtn');
    const settingsBtn = document.getElementById('settingsBtn');
    const closeSettingsBtn = document.getElementById('closeSettingsBtn');
    const themeBtn = document.getElementById('themeBtn');

    const questionEl = document.getElementById('question');
    const optionsEl = document.getElementById('options');
    const feedbackEl = document.getElementById('feedback');
    const questionCounterEl = document.getElementById('questionCounter');
    const progressFillEl = document.getElementById('progressFill');
    const quizTitleEl = document.getElementById('quizTitle');
    const finalScoreEl = document.getElementById('finalScore');
    const resultsTextEl = document.getElementById('resultsText');

    const studyModeToggle = document.getElementById('studyModeToggle');
    const studyModeText = document.getElementById('studyModeText');
    const darkModeToggle = document.getElementById('darkModeToggle');
    const darkModeText = document.getElementById('darkModeText');
    const modeIndicator = document.getElementById('modeIndicator');
    const quizModeIndicator = document.getElementById('quizModeIndicator');

    let currentQuestions = [];
    let wrongAnswers = [];
    let spacedRepetitionQueue = []; // For questions that need to be repeated
    let currentQuestionIndex = 0;
    let score = 0;
    let answered = false;
    let isReviewing = false;

    // Theme handlers
    themeBtn.addEventListener('click', toggleDarkMode);

    darkModeToggle.addEventListener('change', (e) => {
        isDarkMode = e.target.checked;
        applyTheme();
    });

    function toggleDarkMode() {
        isDarkMode = !isDarkMode;
        darkModeToggle.checked = isDarkMode;
        applyTheme();
    }

    function applyTheme() {
        if (isDarkMode) {
            document.documentElement.setAttribute('data-theme', 'dark');
            themeBtn.textContent = '☀️';
            themeBtn.title = 'Switch to light mode';
            darkModeText.textContent = 'Dark Mode';
        } else {
            document.documentElement.setAttribute('data-theme', 'light');
            themeBtn.textContent = '🌙';
            themeBtn.title = 'Switch to dark mode';
            darkModeText.textContent = 'Light Mode';
        }
    }

    // Settings handlers
    settingsBtn.addEventListener('click', () => {
        settingsOverlay.style.display = 'flex';
    });

    closeSettingsBtn.addEventListener('click', () => {
        settingsOverlay.style.display = 'none';
    });

    settingsOverlay.addEventListener('click', (e) => {
        if (e.target === settingsOverlay) {
            settingsOverlay.style.display = 'none';
        }
    });

    studyModeToggle.addEventListener('change', (e) => {
        isSpacedRepetition = e.target.checked;
        const modeText = isSpacedRepetition ? 'Spaced Repetition Mode' : 'Elimination Mode';
        studyModeText.textContent = modeText;
        modeIndicator.textContent = modeText + ' Active';
        updateModeIndicators();
    });

    function updateModeIndicators() {
        const modeText = isSpacedRepetition ? 'Spaced Repetition' : 'Elimination Mode';
        modeIndicator.textContent = modeText + ' Active';
        quizModeIndicator.textContent = modeText;
    }

    // Quiz logic
    startBtn.addEventListener('click', startQuiz);
    nextBtn.addEventListener('click', nextQuestion);
    reviewBtn.addEventListener('click', startReview);
    restartBtn.addEventListener('click', restartQuiz);

    function startQuiz() {
        isReviewing = false;
        wrongAnswers = [];
        spacedRepetitionQueue = [];
        currentQuestions = [...allQuestions];
        quizTitleEl.textContent = 'Biology Quiz';
        commonStart();
    }

    function startReview() {
        isReviewing = true;
        currentQuestions = [...wrongAnswers];
        wrongAnswers = [];
        spacedRepetitionQueue = [];
        quizTitleEl.textContent = 'Reviewing Mistakes';
        commonStart();
    }

    function commonStart() {
        currentQuestionIndex = 0;
        score = 0;
        shuffleArray(currentQuestions);
        startScreen.style.display = 'none';
        resultsScreen.style.display = 'none';
        quizScreen.style.display = 'block';
        updateModeIndicators();
        showQuestion();
    }

    function shuffleArray(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
    }

    function showQuestion() {
        answered = false;
        const q = currentQuestions[currentQuestionIndex];
        const allAnswers = [q.correct, ...q.wrong];
        shuffleArray(allAnswers);

        questionEl.textContent = q.question;

        let totalQuestions = currentQuestions.length;
        if (isSpacedRepetition && spacedRepetitionQueue.length > 0) {
            totalQuestions += spacedRepetitionQueue.length;
        }

        questionCounterEl.textContent = `Question ${currentQuestionIndex + 1} of ${totalQuestions}`;

        const progress = ((currentQuestionIndex) / totalQuestions) * 100;
        progressFillEl.style.width = progress + '%';

        optionsEl.innerHTML = '';
        allAnswers.forEach(answer => {
            const option = document.createElement('div');
            option.className = 'option';
            option.textContent = answer;
            option.addEventListener('click', () => selectAnswer(option, answer, q.correct));
            optionsEl.appendChild(option);
        });

        feedbackEl.style.display = 'none';
        nextBtn.disabled = true;
    }

    function selectAnswer(optionElement, answer, correctAnswer) {
        if (answered) return;
        answered = true;

        const allOptions = document.querySelectorAll('.option');
        allOptions.forEach(opt => {
            opt.classList.add('disabled');
            if (opt.textContent === correctAnswer) {
                opt.classList.add('correct');
            }
        });

        if (answer === correctAnswer) {
            score++;
            feedbackEl.className = 'feedback correct';
            feedbackEl.textContent = 'Correct! Well done.';
        } else {
            optionElement.classList.add('incorrect');

            if (isSpacedRepetition) {
                // Add to spaced repetition queue for later review
                spacedRepetitionQueue.push(currentQuestions[currentQuestionIndex]);
            } else {
                // Elimination mode - add to wrong answers for final review
                wrongAnswers.push(currentQuestions[currentQuestionIndex]);
            }

            feedbackEl.className = 'feedback incorrect';
            feedbackEl.textContent = `Incorrect. The correct answer is: ${correctAnswer}`;
        }

        feedbackEl.style.display = 'block';
        nextBtn.disabled = false;
    }

    function nextQuestion() {
        currentQuestionIndex++;

        // Check if we've finished the current set of questions
        if (currentQuestionIndex >= currentQuestions.length) {
            if (isSpacedRepetition && spacedRepetitionQueue.length > 0) {
                // Move spaced repetition questions back to main queue
                currentQuestions = [...spacedRepetitionQueue];
                spacedRepetitionQueue = [];
                currentQuestionIndex = 0;
                shuffleArray(currentQuestions);
                showQuestion();
            } else {
                showResults();
            }
        } else {
            showQuestion();
        }
    }

    function showResults() {
        quizScreen.style.display = 'none';
        resultsScreen.style.display = 'block';

        const totalQuestions = isReviewing ? wrongAnswers.length : allQuestions.length;
        const percentage = Math.round((score / currentQuestions.length) * 100);
        finalScoreEl.textContent = `${percentage}%`;
        resultsTextEl.textContent = `You got ${score} out of ${currentQuestions.length} questions correct!`;

        // Show review button if there are wrong answers (elimination mode only)
        reviewBtn.style.display = (!isSpacedRepetition && wrongAnswers.length > 0) ? 'inline-block' : 'none';
    }

    // Initialize theme on page load
    applyTheme();

    function restartQuiz() {
        resultsScreen.style.display = 'none';
        startScreen.style.display = 'block';
    }
});
